const express = require('express'); const mongoose = require('mongoose'); const session = require('express-session'); const bcrypt = require('bcryptjs'); const path = require('path'); const app = express(); const PORT = process.env.PORT || 3000; // Middleware app.use(express.json()); app.use(express.urlencoded({ extended: true })); app.use(express.static(path.join(__dirname, 'public'))); app.use(session({ secret: 'genz_bot_super_secret_key_2026', resave: false, saveUninitialized: false, cookie: { maxAge: 24 * 60 * 60 * 1000 } })); // MongoDB Atlas Database Connection const MONGO_URI = process.env.MONGO_URI || 'YOUR_MONGODB_ATLAS_URI'; mongoose.connect(MONGO_URI) .then(async () => { console.log('MongoDB Atlas Connected Successfully!'); // Initialize default admin if not exists const adminCount = await Admin.countDocuments(); if (adminCount === 0) { const hashedPassword = await bcrypt.hash('admin12345', 10); await Admin.create({ username: 'admin', password: hashedPassword }); console.log('Default Admin Account Created (username: admin, pass: admin12345)'); } }) .catch(err => console.error('MongoDB Connection Error:', err)); // ==================== DATABASE SCHEMAS ==================== const AdminSchema = new mongoose.Schema({ username: { type: String, required: true, unique: true, default: 'admin' }, password: { type: String, required: true } }); const Admin = mongoose.model('Admin', AdminSchema); const UserSchema = new mongoose.Schema({ email: { type: String, required: true, unique: true }, password: { type: String, required: true }, activeDeviceId: { type: String, default: null }, sessionToken: { type: String, default: null }, planExpiresAt: { type: Date, default: null }, createdAt: { type: Date, default: Date.now } }); const User = mongoose.model('User', UserSchema); const KeySchema = new mongoose.Schema({ key: { type: String, required: true, unique: true }, durationDays: { type: Number, required: true }, targetEmail: { type: String, default: null }, // Dedicated User Email isUsed: { type: Boolean, default: false }, usedBy: { type: String, default: null }, createdAt: { type: Date, default: Date.now } }); const Key = mongoose.model('Key', KeySchema); // ==================== MIDDLEWARES ==================== function isAdmin(req, res, next) { if (req.session && req.session.isAdmin) { return next(); } return res.status(403).json({ success: false, message: 'Access Denied: Admin privileges required.' }); } function generateActivationKey(length = 16) { const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'; let key = ''; for (let i = 0; i < length; i++) { if (i > 0 && i % 4 === 0) key += '-'; key += chars.charAt(Math.floor(Math.random() * chars.length)); } return key; } // ==================== ADMIN API ROUTES ==================== // 1. Admin Login API app.post('/api/admin/login', async (req, res) => { try { const { username, password } = req.body; const adminDoc = await Admin.findOne({ username }); if (!adminDoc) { return res.json({ success: false, message: 'Invalid Admin Credentials!' }); } const isMatch = await bcrypt.compare(password, adminDoc.password); if (isMatch) { req.session.isAdmin = true; return res.json({ success: true, message: 'Admin authenticated successfully!' }); } return res.json({ success: false, message: 'Invalid Admin Credentials!' }); } catch (err) { res.status(500).json({ success: false, message: err.message }); } }); // 2. Change Admin Password API app.post('/api/admin/change-password', isAdmin, async (req, res) => { try { const { currentPassword, newPassword } = req.body; const adminDoc = await Admin.findOne(); const isMatch = await bcrypt.compare(currentPassword, adminDoc.password); if (!isMatch) { return res.json({ success: false, message: 'Current password is incorrect!' }); } const hashedNewPassword = await bcrypt.hash(newPassword, 10); adminDoc.password = hashedNewPassword; await adminDoc.save(); res.json({ success: true, message: 'Admin password updated successfully!' }); } catch (err) { res.status(500).json({ success: false, message: err.message }); } }); // 3. Generate Key with Target Email Option app.post('/api/admin/generate-key', isAdmin, async (req, res) => { try { const { durationDays, targetEmail } = req.body; const newKey = generateActivationKey(); const keyDoc = new Key({ key: newKey, durationDays: parseInt(durationDays) || 30, targetEmail: targetEmail ? targetEmail.trim().toLowerCase() : null }); await keyDoc.save(); res.json({ success: true, key: newKey, durationDays: keyDoc.durationDays, targetEmail: keyDoc.targetEmail }); } catch (err) { res.status(500).json({ success: false, message: err.message }); } }); // 4. Get Admin Dashboard Data app.get('/api/admin/dashboard-data', isAdmin, async (req, res) => { try { const keys = await Key.find().sort({ createdAt: -1 }); const users = await User.find().select('-password').sort({ createdAt: -1 }); res.json({ success: true, keys, users }); } catch (err) { res.status(500).json({ success: false, message: err.message }); } }); // 5. User Management Actions app.post('/api/admin/user-action', isAdmin, async (req, res) => { try { const { userId, action } = req.body; if (action === 'delete') { await User.findByIdAndDelete(userId); return res.json({ success: true, message: 'User account removed.' }); } else if (action === 'unlock_device') { await User.findByIdAndUpdate(userId, { activeDeviceId: null, sessionToken: null }); return res.json({ success: true, message: 'Device lock reset successfully.' }); } } catch (err) { res.status(500).json({ success: false, message: err.message }); } }); // Routes app.get('/admin', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'admin.html')); }); app.listen(PORT, () => console.log(`Server live on port ${PORT}`));
